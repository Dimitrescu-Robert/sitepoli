\section*{Partea I: Enunțuri}

\section*{Partea II: Rezolvări}

\begin{enumerate}
    \item 
    $$12960\frac{\text{km}}{\text{h}^2}=12960\cdot\frac{1000 \text{ m}}{3600^2 \text{ s}^2}=1 \text{ m/s}^2$$
    Răspuns corect: \textbf{E/D}.

    \item 
    Fie $N_d = $ numărul de molecule disociate și $N_i = $ numărul inițial de molecule
    $$N_d = \alpha N_i$$
    $$N_{\text{atomi}} = 2N_d = 2\alpha N_i$$
    $$N_{nd} = \text{numărul de molecule nedisociate} = N_i - N_d = N_i - \alpha N_i = N_i(1-\alpha)$$
    $$N_t = \text{numărul final de molecule} = N_{\text{atomi}} + N_{nd} = 2\alpha N_i + N_i(1-\alpha) = N_i(1+\alpha)$$
    $$N_t = N_i(1+\alpha) \cdot \frac{1}{N_a} \implies \nu_t = \nu(1+\alpha)$$
    $$P = \frac{\nu(1+\alpha)RT}{V} \approx 90,8 \cdot 10^3 \text{ Pa}$$
    Răspuns corect: \textbf{A}.

    \item 
    Fie $I$ punctul în care se întâlnesc cele două vapoare la ora 14.
    M.R.U. $\implies v = \frac{d}{t} \implies d = v \cdot t$
    Fie $\tau$ - timpul parcurs de cele două vapoare până la întâlnire și d - distanța de la A la B, ecuațiile mișcării sunt:
    \begin{equation*}
        \begin{cases}
            \tau(v_1+v_2)=d \\
            v_1(4+\tau)=d \\
            v_2(9+\tau)=d
        \end{cases}
    \end{equation*}
    $$\tau \cdot v_1 + \tau \cdot v_2 = v_1(\tau+4) \implies \tau \cdot v_2 = 4v_1 \quad \text{(I)}$$
    $$v_1(4+\tau) = v_2(9+\tau) \implies v_1 = v_2\frac{\tau+9}{\tau+4} \quad \text{(II)}$$
    Din (I) și (II) obținem că $\tau(\tau+4) = 4(\tau+9) \implies \tau = 6\text{h}$
    Ora de plecare $= 14 - 6 = 8$
    Răspuns corect: \textbf{C}.

    \item 
    Fie T tensiunea în fir.
    Corpurile se deplasează pe roți, deci nu există frecare!
    $M_2$ se află în repaus $\iff T = G_2 \implies T = gM_2$
    Cum cele două corpuri nu se mișcă față de M $\implies$ față de sol, pe orizontală corpurile se mișcă cu accelerația a imprimată de F corpului M. Principiul II pe orizontală pentru corpul $M_1$ este: $T = aM_1$
    Din cele două formule ale tensiunii în fir $\implies a = g\frac{M_2}{M_1}$
    $$F = (M+M_1+M_2)a = (M+M_1+M_2)g\frac{M_2}{M_1} = 612 \text{ N}$$
    Răspuns corect: \textbf{B}.

    \item 
    Atunci când cilindrul 1 este încălzit, are loc o transformare generală:
    $$\frac{P_0 V_1}{T} = \frac{P_1 V'_1}{T_1}$$ unde $V'_1$ este volumul după încălzire, iar $V'_1 = V_1 + xS_1$
    Gazul din cilindrul 2 suferă o transformare izotermă:
    $$P_0 V_2 = P_2 V'_2$$ unde $V'_2$ este noul volum al cilindrului, iar $V'_2 = V_2 - xS_2$
    Pistoanele se află în echilibru dacă forțele ce acționează asupra lor se află în echilibru:
    $$(P_1 - P_0)S_1 = (P_2 - P_0)S_2$$
    Înlocuind volumele noi și împreună cu cea de-a treia ecuație se formează un sistem de trei ecuații cu trei necunoscute pe care îl rezolvăm:
    \begin{equation*}
        \begin{cases}
            \frac{P_0 V_1}{T} = \frac{P_1(V_1+xS_1)}{T_1} \\
            P_0 V_2 = P_2(V_2-xS_2) \\
            (P_1-P_0)S_1 = (P_2-P_0)S_2
        \end{cases}
    \end{equation*}
    Rezolvând sistemul obținem $P_1 = 1,2 \cdot 10^5 \text{ Pa}$, $P_2 = 1,6 \cdot 10^6 \text{ Pa}$ și $x = 0,075 \text{ m}$
    $$x = 7,5 \text{ cm}$$
    Răspuns corect: \textbf{D}.

    \item 
    $$P_b = U_b I, \quad P_L = I^2 R_1, \quad P_{ext} = P_b + P_L$$
    Obținem o ecuație de gradul 2 în I: $I^2 + 3I - 10 = 0 \implies I = 2 \text{ A}$
    $$R_b = 7,5 \, \Omega$$
    $$P_{totala} = P_{ext} + P_{int}$$
    $$P_b = U_b I = 30 \text{ W} \implies P_{totala} = 60 \text{ W}$$
    $$P_{int} = 10 \text{ W}$$
    $$P_{int} = I^2 r \implies r = 2,5 \, \Omega$$
    $$U = E - Ir \implies E = U + Ir; \quad U = I R_e$$
    $$U = 2(7,5 + 5) = 25 \text{ V}$$
    $$E = 30 \text{ V}$$
    $$I_{sc} = \frac{E}{r} = 12 \text{ A}$$
    Răspuns corect: \textbf{A}.

    \item 
    Din $R = R_0(1+\alpha t)$, obținem $R_1 = R_0(1+\alpha t_1)$ și $R_2 = R_0(1+\alpha t_2)$
    Facem raportul și obținem:
    $$\frac{R_2}{R_1} = \frac{1+\alpha t_1}{1+\alpha t_2} = n$$
    apoi $\alpha = \frac{n-1}{t_2-nt_1}$, deci răspunsul corect este F.

    \item 
    Notăm cu R rezistența echivalentă a circuitului format din n celule. Dacă se mai adaugă o celulă, rezistența echivalentă trebuie să fie egală tot cu R.
    $$R = R_1 + \frac{R_2 R}{R+R_2} \implies R^2 - RR_1 - R_1 R_2 = 0$$
    Cu soluția pozitivă: $R = \frac{R_1+\sqrt{R_1^2+4R_1R_2}}{2} \implies R = 1+\sqrt{3} \, \Omega$
    Răspuns corect: \textbf{B}.

    \item 
    $$\eta = 1 - \frac{|Q_c|}{Q_p} \implies Q_p = \frac{Q_c}{1-\eta} = 300 \text{ J}$$
    $$L = Q_p - |Q_c| = 90 \text{ J}$$
    $$P = nL = 900 \text{ W}$$
    Răspuns corect: \textbf{A}.

    \item 
    Fie v = viteza corpului la nivelul solului, $v_0=0$, la nivelul solului energia potențială este nulă.
    Se conservă energia mecanică:
    $$mgR = \frac{mv^2}{2} \implies v = \sqrt{2gR} = 10 \text{ m/s}$$
    Răspuns corect: \textbf{B}.

\end{enumerate}
