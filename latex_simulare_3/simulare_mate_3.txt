\section*{Set de 10 Probleme - Matematică}

\begin{enumerate}
    % Problema 1 
    \item Fie funcția $f(x) = \{x\}(1-\{x\})$, unde $\{x\}$ reprezintă partea fracționară a lui $x$. Calculați:
    \[ \int_{0}^{10} f(x) \, dx \]
    \begin{enumerate}[label=\textbf{\Alph*)}]
        \item $\frac{5}{3}$
        \item $\frac{10}{3}$
        \item $\frac{1}{6}$
        \item $5$
        \item $0$
        \item $1$
    \end{enumerate}

    % Problema 2 [cite: 2]
    \item Rezolvați în mulțimea numerelor reale sistemul de ecuații:
    \[
    \begin{cases} 
    x + y + z = 3 \\ 
    x^2 + y^2 + z^2 = 3 \\ 
    x^3 + y^3 + z^3 = 3 
    \end{cases}
    \]
    \begin{enumerate}[label=\textbf{\Alph*)}]
        \item $(1, 1, 1)$
        \item $(0, 1, 2)$
        \item $(3, 0, 0)$
        \item $\emptyset$
        \item $(1, 2, 0)$
        \item $(-1, 2, 2)$
    \end{enumerate}

    % Problema 3 
    \item Fie funcția $f(x) = \arctan(x) + \arctan\left(\frac{1-x}{1+x}\right)$. Calculați valoarea derivatei $f'(2026)$.
    \begin{enumerate}[label=\textbf{\Alph*)}]
        \item $0$
        \item $1$
        \item $\frac{1}{2026}$
        \item $\frac{\pi}{4}$
        \item $\frac{1}{1+2026^2}$
        \item $-1$
    \end{enumerate}

    % Problema 4 [cite: 4]
    \item Determinați numerele reale $a, b, c$ astfel încât graficul funcției $f(x) = ax^3 + bx^2 + c$ să fie simetric față de punctul $M(1,2)$.
    \begin{enumerate}[label=\textbf{\Alph*)}]
        \item $a \in \mathbb{R}^*, b=-3a, c=2+2a$
        \item $a=1, b=3, c=1$
        \item $a=1, b=-3, c=0$
        \item $a=0, b=0, c=2$
        \item $a=1, b=1, c=0$
        \item $a, b, c \in \mathbb{R}$
    \end{enumerate}

    % Problema 5 
    \item Aflați numărul punctelor de extrem local ale funcției $f : \mathbb{R} \to \mathbb{R}$, definită prin:
    \[ f(x) = |x^2 - 4x + 3| \cdot e^{-x} \]
    \begin{enumerate}[label=\textbf{\Alph*)}]
        \item $4$
        \item $2$
        \item $3$
        \item $1$
        \item $5$
        \item $0$
    \end{enumerate}

    % Problema 6 [cite: 6]
    \item Rezolvați ecuația trigonometrică:
    \[ \cos(2x) + 3\sin(x) - 2 = 0, \quad x \in [0, 2\pi] \]
    \begin{enumerate}[label=\textbf{\Alph*)}]
        \item $\{\frac{\pi}{6}, \frac{\pi}{2}, \frac{5\pi}{6}\}$
        \item $\{\frac{\pi}{6}, \frac{5\pi}{6}\}$
        \item $\{\frac{\pi}{2}\}$
        \item $\{0, \pi, 2\pi\}$
        \item $\{\frac{\pi}{3}, \frac{\pi}{2}, \frac{2\pi}{3}\}$
        \item $\emptyset$
    \end{enumerate}

    % Problema 7 [cite: 6, 7]
    \item Aflați formula pentru suma $S_n$, unde:
    \[ S_n = \sum_{k=1}^{n} k \cdot k! \]
    \begin{enumerate}[label=\textbf{\Alph*)}]
        \item $(n+1)! - 1$
        \item $n!$
        \item $(n+1)!$
        \item $n \cdot n!$
        \item $\frac{n(n+1)}{2}$
        \item $(n-1)!$
    \end{enumerate}

    % Problema 8 [cite: 8, 9]
    \item Fie matricea $A = \frac{1}{2} \begin{pmatrix} 2 & 3 & 2 \\ 3 & 5 & 4 \\ 4 & 7 & 8 \end{pmatrix}$. Se știe că $A \cdot B \cdot A = A^{-1}$. Calculați $\det(B^T)$.
    \begin{enumerate}[label=\textbf{\Alph*)}]
      \item $64$
      \item $\dfrac{1}{64}$
      \item $8$
      \item $\dfrac{1}{8}$
      \item $-64$
      \item $1$
    \end{enumerate}

    % Problema 9 [cite: 13]
    \item Să se determine mulțimea soluțiilor ecuației:
    \[ \frac{-z^3-1}{z+1} = z-1, \quad z \in \mathbb{C} \setminus \{-1\} \]
    \begin{enumerate}[label=\textbf{\Alph*)}]
        \item $\{0\}$
        \item $\{0, 1\}$
        \item $\{-1\}$
        \item $\{0, -1\}$
        \item $\{-1, 1\}$
        \item $\emptyset$
    \end{enumerate}

    % Problema 10 [cite: 17, 18]
    \item Fie $a, b \in (0, \infty)$ și $f_k(x) = k^x$. Calculați valoarea limitei:
    \[ L = \lim_{x \to 0} \left( \frac{f_a(x) + f_b(x)}{2} \right)^{\frac{1}{x}} \]
    \begin{enumerate}[label=\textbf{\Alph*)}]
        \item $\sqrt{ab}$
        \item $\frac{a+b}{2}$
        \item $ab$
        \item $1$
        \item $e^{ab}$
        \item $\infty$
    \end{enumerate}

\end{enumerate}
